% ─────────────────────────────────────────────────────────────────────────────
% Figure 1 — P7 Workflow Schematic  (QMSE Phase D architecture)
%
% Pipeline: ANP Library → SMILES → QMSE encoding (BondOrderMatrix /
%           CoulombMatrix) → 8-qubit parameterised QMSE circuit (PennyLane)
%           → TF Dense head → Activity prediction
%
% Three evaluation phases shown as a row at the bottom:
%   Phase B smoke-test (P1 Set A, LOO-CV)
%   Phase C/D canonical benchmark (P3 sub n=1000, 5-fold CV)
%   Stereo probe (Qiskit Aer, n_qubits=n_atoms)
%
% Rewritten 14 October 2026 for QMSE (Phase D) refocus.
% ─────────────────────────────────────────────────────────────────────────────

% Colour palette (colour-blind safe, consistent with matplotlib figures)
\definecolor{colANP}    {RGB}{33,102,172}   % blue   — data/input
\definecolor{colEnc}    {RGB}{140,81,10}    % brown  — encoding step
\definecolor{colQML}    {RGB}{214,96,77}    % red    — quantum block
\definecolor{colTF}     {RGB}{77,146,33}    % green  — TF classical head
\definecolor{colOut}    {RGB}{35,139,69}    % green  — output
\definecolor{colPhase}  {RGB}{117,112,179}  % purple — evaluation phases
\definecolor{colBg}     {RGB}{247,247,247}  % light grey background

\tikzset{
  % ── Main pipeline node styles ──────────────────────────────────────────────
  datanode/.style={

  },
  qnode/.style={
    rectangle, rounded corners=3pt,
    draw=colQML!80, fill=colQML!12, thick,
    text width=2.4cm, align=center,
    font=\small\sffamily, inner sep=5pt, minimum height=1.0cm
  },
  tfnode/.style={
    rectangle, rounded corners=3pt,
    draw=colTF!80, fill=colTF!10, thick,

  },
  outnode/.style={
    rectangle, rounded corners=5pt,

  % ── Arrows ─────────────────────────────────────────────────────────────────
  mainarrow/.style={
    -{Stealth[length=5pt,width=4pt]},
    thick, color=black!55
  },
  phasearrow/.style={
    -{Stealth[length=4pt,width=3pt]},
    thick, color=colPhase!70, dashed
  },

    font=\scriptsize\sffamily, color=black!55,
    fill=white, inner sep=1.5pt
  }
}

\begin{tikzpicture}[node distance=0.55cm and 0.45cm]

% ── Row 1: main pipeline ──────────────────────────────────────────────────────

\node[datanode] (anp) {
  \textbf{ANP Library}\\[2pt]
  {\footnotesize SMILES}
};

\node[encnode, right=1.1cm of anp] (qmse) {
  \textbf{QMSE}\\
  \textbf{Encoding}\\[2pt]
  {\footnotesize BondOrderMatrix\\or CoulombMatrix\\$n\!\times\!n$ matrix}
};

\node[qnode, right=1.0cm of qmse] (circuit) {
  \textbf{QMSE Circuit}\\[2pt]
  {\footnotesize $n_q\!=\!8$ qubits\\depth 2, RYY\\PennyLane}
};

\node[tfnode, right=1.0cm of circuit] (tfhead) {
  \textbf{TF Dense}\\
  \textbf{Head}\\[2pt]
  {\footnotesize Dense(128)$\times$3\\BatchNorm\\Dropout}
};

\node[outnode, right=1.0cm of tfhead] (pred) {
  \textbf{Activity}\\
  \textbf{Score}\\[2pt]
  {\footnotesize $\hat{y}\in[0,1]$}
};

% Main pipeline arrows
\draw[mainarrow] (anp)     -- node[arrowlabel, above] {SMILES}             (qmse);
\draw[mainarrow] (qmse)    -- node[arrowlabel, above] {angles}             (circuit);
\draw[mainarrow] (circuit) -- node[arrowlabel, above] {$\langle Z_q\rangle$} (tfhead);
\draw[mainarrow] (tfhead)  -- node[arrowlabel, above] {}                   (pred);

% ── Circuit encoding detail (zoom box below QMSE + circuit) ──────────────────

\node[below=1.35cm of qmse, xshift=1.5cm,
      draw=colEnc!55, fill=colEnc!5,
      rounded corners=3pt, inner sep=7pt,
      text width=5.2cm, align=left,
      font=\scriptsize\sffamily] (zoom) {
  \textbf{Encoding angles (8-qubit register):}\\[3pt]
  Single-qubit: $\phi_i = \mathbf{M}_{ii}$\quad(diagonal)\\[1pt]
  Two-qubit:\quad $\phi_{ij} = \mathbf{M}_{ij}$\quad(off-diagonal)\\[4pt]
  \textbf{Circuit (depth 2):}\\[2pt]
  $|0\rangle\!-\!R_X(\phi_0)\!-\!\mathrm{RYY}(\phi_{01})\!-\!\cdots$\\[1pt]
  $|0\rangle\!-\!R_X(\phi_1)\!-\!\mathrm{RYY}(\phi_{12})\!-\!\cdots$\\[1pt]
  $\;\vdots$\\[1pt]
  $|0\rangle\!-\!R_X(\phi_7)\!-\!\cdots$\\[3pt]
  {\color{black!50} Stereo:\ S-centre $\Rightarrow$
    $\mathbf{M}_{ii}\!<\!0$;\quad
    Z-bond $\Rightarrow$ $\mathbf{M}_{ij}\!<\!0$}
};

% Connector from circuit node to zoom box
\draw[phasearrow, color=colEnc!55]
  (circuit.south) -- ++(0,-0.3) -| (zoom.north);

% ── Row 3: evaluation phases ─────────────────────────────────────────────────

\node[phasenode, below=5.0cm of anp] (pb) {

};

\node[phasenode, right=0.5cm of pb] (pcd) {
